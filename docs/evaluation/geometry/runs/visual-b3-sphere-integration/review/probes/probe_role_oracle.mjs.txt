const TERRACOTTA = [200, 90, 50], COBALT = [37, 99, 235], BG = [245, 244, 240], GREY = [150, 150, 150];

const edges = poly.map((a, i) => ({ a, b: poly[(i + 1) % 4], hidden: false }));

function img() {
  const rgba = new Uint8Array(W * H * 4);
  const put = (x, y, c) => { if (x < 0 || y < 0 || x >= W || y >= H) return; rgba.set([...c, 255], (y * W + x) * 4); };
  for (let y = 0; y < H; y++) for (let x = 0; x < W; x++) put(x, y, BG);
  const rect = (x0, y0, x1, y1, c) => { for (let y = y0; y < y1; y++) for (let x = x0; x < x1; x++) put(x, y, c); };
  const line = (a, b, c, t0 = 0, t1 = 1, w = 1) => { for (let t = t0; t <= t1; t += 0.002) { const x = Math.round(a.x + (b.x - a.x) * t), y = Math.round(a.y + (b.y - a.y) * t); for (let d = -w; d <= w; d++) { put(x + d, y, c); put(x, y + d, c); } } };
  return { w: W, h: H, rgba, put, rect, line };
}

  console.log(JSON.stringify({ name, pass: r.pass, reasons: r.reasons, blue: r.blue, cov: r.edges.map((e) => e.coverage) }));
};
