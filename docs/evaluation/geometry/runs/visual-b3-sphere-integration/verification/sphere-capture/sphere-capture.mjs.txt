// Capture the offline sphere sample at the final step (desktop 1440 / mobile 390). Usage: node sphere-capture.mjs <port> <outDir>
import { chromium } from '../frontend/node_modules/playwright/index.mjs';
import { readFileSync, mkdirSync, writeFileSync } from 'node:fs';
import { createHash } from 'node:crypto';

const [port = '3141', OUT = '.superpowers/sphere-capture'] = process.argv.slice(2);
mkdirSync(OUT, { recursive: true });
const ids = (process.env.SAMPLE_IDS || 'cau-cat-mat-phang').split(',');
const samples = JSON.parse(readFileSync('frontend/src/data/geometry-samples.json', 'utf8')).samples;
const b = await chromium.launch({ channel: 'chrome', headless: true, args: ['--use-angle=swiftshader', '--enable-unsafe-swiftshader'] });
const rows = [], errors = [];
try {
  for (const width of [1440, 390]) {
    const ctx = await b.newContext({ viewport: { width, height: width < 600 ? 844 : 900 }, deviceScaleFactor: width < 600 ? 2 : 1 });
    await ctx.route('**/*', (r) => {
      const u = new URL(r.request().url());
      if (u.hostname !== '127.0.0.1') return r.abort();
      if (u.pathname.startsWith('/api/')) return r.fulfill({ json: u.pathname === '/api/health' ? { ok: true, hasKey: false } : { user: null } });
      return r.continue();
    });
    const p = await ctx.newPage();
    p.on('pageerror', (e) => errors.push(e.message.split('\n')[0]));
    for (const id of ids) {
      await p.goto(`http://127.0.0.1:${port}`);
      await p.evaluate(async (env) => { const { useAppStore } = await import('/src/state/store.ts'); useAppStore.getState().loadEnvelope(env); },
        samples.find((s) => s.id === id).envelope);
      await p.locator('.geo3d-canvas canvas').waitFor();
      await p.locator('.geo3d-scrub input').evaluate((el) => {
        const set = Object.getOwnPropertyDescriptor(HTMLInputElement.prototype, 'value').set;
        set.call(el, el.max); el.dispatchEvent(new Event('input', { bubbles: true })); el.dispatchEvent(new Event('change', { bubbles: true }));
      });
      await p.waitForTimeout(900);
      const row = { id, width, images: [] };
      const shot = async (state) => {
        const bytes = await p.screenshot({ fullPage: true }), file = `${id}-${width}-${state}.png`;
        writeFileSync(`${OUT}/${file}`, bytes);
        row.images.push({ state, file, sha256: createHash('sha256').update(bytes).digest('hex') });
      };
      row.canvas_labels = await p.locator('.geo3d-label').allTextContents();
      await shot('final-mode-a');
      row.mode_a_text_has_answer = /16π|\b4\b/.test(await p.locator('.geo3d').innerText());
      await p.getByRole('button', { name: 'Phân tích toán học', exact: true }).click();
      await p.waitForTimeout(500);
      await shot('analysis-open');
      const txt = await p.locator('body').innerText();
      row.suspicious_text = [...new Set(txt.split('\n').filter((l) => /đại lượng|đi qua|Điểm [AHUW]\b/i.test(l)).map((l) => l.trim()))];
      await p.getByRole('button', { name: 'Phân tích toán học', exact: true }).click();
      await p.waitForTimeout(300);
      const box = await p.locator('.geo3d-canvas canvas').boundingBox();
      await p.mouse.click(box.x + box.width * 0.5, box.y + box.height * 0.72);
      await p.waitForTimeout(600);
      row.selected_after_click = await p.evaluate(() => window.__geo3d_selected_id ?? null);
      await shot('click-lower-sphere');
      const ins = await p.locator('body').innerText();
      row.inspector_suspicious = [...new Set(ins.split('\n').filter((l) => /đại lượng|đi qua/i.test(l)).map((l) => l.trim()))];
      await p.mouse.click(box.x + 4, box.y + box.height - 4);
      await p.getByRole('button', { name: 'Hiển thị', exact: true }).click();
      const aux = p.getByRole('menuitemcheckbox').filter({ hasText: 'Hình phụ' });
      if (await aux.count()) await aux.click();
      await p.waitForTimeout(500);
      await shot('auxiliary-on');
      row.canvas_labels_aux = await p.locator('.geo3d-label').allTextContents();
      rows.push(row);
    }
    await ctx.close();
  }
} finally { await b.close(); }
writeFileSync(`${OUT}/receipt.json`, JSON.stringify({ provider_calls: 0, rows, errors }, null, 2));
console.log(JSON.stringify({ rows: rows.map((r) => ({ id: r.id, width: r.width, labels: r.canvas_labels, suspicious: r.suspicious_text, modeA: r.mode_a_text_has_answer })), errors }, null, 1));
