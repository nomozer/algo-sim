// Acceptance for the Vite dev-server EPERM fix. Usage (from the checkout root): node .superpowers/dev-server-acceptance.mjs <out.json>
// Each round: spawn `npm run dev` (optionally `-- --force`), wait for "ready", load the app in Chrome (forces dependency
// optimisation), edit a stylesheet with a harmless comment and wait for the browser's "[vite] hot updated" message,
// restore the file byte-for-byte, stop ONLY the spawned process tree. Fails on any EPERM line.
import { spawn, execFileSync } from "node:child_process";
import { readFileSync, writeFileSync } from "node:fs";
import { createRequire } from "node:module";

const require = createRequire(new URL("../frontend/package.json", import.meta.url));
const { chromium } = require("playwright");
const OUT = process.argv[2];
const CSS = new URL("../frontend/src/styles/global.css", import.meta.url);
const original = readFileSync(CSS);
const strip = (s) => s.replace(/\x1b\[[0-9;]*m/g, "");

async function round(name, extra = []) {
  const t0 = Date.now();
  const child = spawn("npm", ["run", "dev", ...(extra.length ? ["--", ...extra] : [])],
    { cwd: new URL("../frontend/", import.meta.url), shell: true, windowsHide: true });
  let log = "";
  child.stdout.on("data", (d) => { log += d; });
  child.stderr.on("data", (d) => { log += d; });
  const exited = new Promise((res) => child.on("exit", (code) => res(code)));
  const row = { name, args: extra };
  try {
    const ready = await Promise.race([
      new Promise((res) => { const iv = setInterval(() => { if (/ready in/.test(strip(log))) { clearInterval(iv); res(true); } }, 100); }),
      exited.then(() => false),
      new Promise((res) => setTimeout(() => res(false), 30000)),
    ]);
    row.ready = ready;
    row.ready_ms = Date.now() - t0;
    if (!ready) return row;
    const browser = await chromium.launch({ channel: "chrome", headless: true, args: ["--use-angle=swiftshader"] });
    try {
      const page = await browser.newPage();
      const consoleLines = [];
      page.on("console", (m) => consoleLines.push(m.text()));
      await page.route("**/api/**", (r) => r.fulfill({ json: r.request().url().includes("/api/health") ? { ok: true, hasKey: false } : null }));
      await page.goto("http://localhost:3000/", { waitUntil: "networkidle", timeout: 60000 });
      row.app_rendered = (await page.locator("#root *").count()) > 0;
      writeFileSync(CSS, Buffer.concat([original, Buffer.from(`\n/* hmr-probe ${Date.now()} */\n`)]));
      const deadline = Date.now() + 15000;
      while (Date.now() < deadline && !consoleLines.some((l) => /\[vite\] (hot updated|css hot updated)/.test(l))) {
        await new Promise((r) => setTimeout(r, 100));
      }
      row.hot_update = consoleLines.find((l) => /hot updated/.test(l)) ?? null;
    } finally {
      writeFileSync(CSS, original);
      await browser.close();
    }
  } finally {
    try { execFileSync("taskkill", ["/PID", String(child.pid), "/T", "/F"], { stdio: "ignore" }); } catch { /* already gone */ }
    row.exit_code = await exited;
    row.eperm = /EPERM/.test(log);
    row.log = strip(log).split(/\r?\n/).filter((l) => l.trim()).slice(0, 25);
  }
  return row;
}

const rows = [];
for (const [name, extra] of [["start-1", []], ["start-2", []], ["force-reoptimise", ["--force"]]]) {
  rows.push(await round(name, extra));
  console.log(JSON.stringify({ name, ready: rows.at(-1).ready, hot: Boolean(rows.at(-1).hot_update), eperm: rows.at(-1).eperm }));
}
const restored = Buffer.compare(readFileSync(CSS), original) === 0;
const pass = restored && rows.every((r) => r.ready && r.app_rendered && r.hot_update && !r.eperm);
writeFileSync(OUT, JSON.stringify({ pass, css_restored_byte_identical: restored, rows }, null, 2));
console.log(JSON.stringify({ pass, restored }));
process.exitCode = pass ? 0 : 1;
