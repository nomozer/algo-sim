def sha(data):return hashlib.sha256(data).hexdigest()
def js(path):return json.loads(path.read_text(encoding='utf-8'))
def write(path,data):
    path.parent.mkdir(parents=True,exist_ok=True)
    path.write_text(data if isinstance(data,str) else json.dumps(data,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
def copy(src,dest):
    data=src.read_bytes();dest.parent.mkdir(parents=True,exist_ok=True)
    if dest.exists():assert dest.read_bytes()==data,f'Artifact changed: {dest}'
    else:dest.write_bytes(data)
    return {'source':str(src.relative_to(ROOT)) if src.is_relative_to(ROOT) else str(src),
            'file':dest.relative_to(STAGE).as_posix(),'bytes':len(data),'sha256':sha(data)}
